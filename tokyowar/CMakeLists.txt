cmake_minimum_required(VERSION 3.16)
project(tokyowar C)

if(NOT CMAKE_C_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "gcc required (the lifted code uses computed labels and statement expressions)")
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
add_compile_options(-fno-strict-aliasing -fwrapv)

# PORTABLE (default ON): pin glibc symbol versions so a binary built here runs on older systems -- the Steam Deck included.
# (Prop Cycle's ../include/glibc_compat.h: one copy for every game.)
option(PORTABLE "Bind glibc symbols to baseline versions (runs on glibc >= 2.34)" ON)
if(PORTABLE AND NOT WIN32 AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  add_compile_options(-include ${CMAKE_CURRENT_SOURCE_DIR}/../include/glibc_compat.h)
endif()

include_directories(include gen)

# THE SHARED ENGINE (../engine): the 68K host runtime (lift_cpu.c) today; the renderer, mixer
# and DSP translation when T4-T6 land (HARD RULE 3 in ../CLAUDE.md).
include(${CMAKE_CURRENT_SOURCE_DIR}/../engine/engine.cmake)

# gen/tw_lifted.c is GENERATED by ../tools/namco22/lift.py from decompiled/pcode.txt:
#   python3 ../tools/namco22/lift.py decompiled/pcode.txt gen tw
# Big goto-heavy C; -O1 keeps the build short and is plenty for a 24 MHz 68EC020.
set_source_files_properties(gen/tw_lifted.c PROPERTIES COMPILE_OPTIONS "-O1;-w")

# The MASTER DSP program (the C71 BIOS + the block the 68K uploads through it), translated to C from
# the ROM files by the shared tool ../tools/gen/c25_translate.py. tools/gen/c25.cov: the addresses
# the interpreter oracle executed (addresses only, no ROM data; committed). The game has no C25
# fetch/decode loop. Block 0x127816 is the master program (FUN_0012ED28 uploads it as block 1).
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_command(OUTPUT ${CMAKE_CURRENT_SOURCE_DIR}/gen/tw_c25.c
  COMMAND ${Python3_EXECUTABLE} ../tools/gen/c25_translate.py --game tw --roms extracted
          --cov tools/gen/c25.cov --out gen/tw_c25.c --func tw_c25_exec
  DEPENDS ../tools/gen/c25_translate.py tools/gen/c25.cov
  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  COMMENT "Translating the master DSP program to C")
set_source_files_properties(gen/tw_c25.c PROPERTIES COMPILE_OPTIONS "-O1;-w")

# The SOUND MCU program (tw1data.8k -- the S22-BIOS ver1.41 and the game's sound program, which also
# reads the cabinet's ports), translated to C from the ROM file by the shared tools/gen/snd_translate.py
# (--game tw). tools/gen/snd.cov: the (address, M/X, length) of every instruction the tw_oracle test
# build executed (addresses only, no ROM data; committed). The game has no M37710 fetch/decode loop.
add_custom_command(OUTPUT ${CMAKE_CURRENT_SOURCE_DIR}/gen/tw_snd_driver.c
  COMMAND ${Python3_EXECUTABLE} ../tools/gen/snd_translate.py tools/gen/snd.cov --game tw --roms extracted
          --out gen/tw_snd_driver.c
  DEPENDS ../tools/gen/snd_translate.py ../tools/gen/snd_rom.py ../tools/gen/snd_spin.inc ../tools/gen/m7700_decode.py tools/gen/snd.cov
  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  COMMENT "Translating the sound MCU program to C")
set_source_files_properties(gen/tw_snd_driver.c PROPERTIES COMPILE_OPTIONS "-O1;-w")

# the game: the 68K (lifted), the master DSP (translated), the sound MCU (translated) and its C352
# the picture and the window: the shared engine's geometry stage, texture bake, GL rasteriser and Super 22 layers
# (ss22_gl.c), around Tokyo Wars' own video RAM (tw_video.c), controls (tw_input.c) and window (tw_host.c)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED sdl2)
find_package(ZLIB REQUIRED)                       # the first-run ROM unpacker (engine/romzip.c)
set(OpenGL_GL_PREFERENCE GLVND)
find_package(OpenGL REQUIRED)
include_directories(${SDL2_INCLUDE_DIRS})
# Windows (MinGW-w64, ../build-windows.sh): the few POSIX calls mapped in ../include/win_compat.h, no console window, and
# everything linked in -- SDL2 statically too -- so TokyoWars.exe needs no DLL beside it. OpenGL is NOT linked: the shared
# engine's gl_dyn.c resolves it at run time, so the bundled Mesa can stand in when Windows has no real driver.
if(WIN32)
  add_compile_options(-include ${CMAKE_CURRENT_SOURCE_DIR}/../include/win_compat.h)
  add_link_options(-mwindows -static-libgcc -static)
  set(SDL2_LIBRARIES ${SDL2_STATIC_LDFLAGS})
  set(TW_GL_LIB "")
  set(TW_GL_SRC ${NAMCO22_ENGINE_DIR}/gl_dyn.c)
else()
  set(TW_GL_LIB OpenGL::GL)
  set(TW_GL_SRC "")
endif()

set(TW_CORE_SRC src/tw_main.c src/tw_mem.c src/tw_hw.c src/tw_env.c src/tw_dsp.c src/tw_snd.c src/tw_rd_stub.c
    src/tw_video.c src/tw_input.c src/tw_host.c src/tw_rom.c
    ${NAMCO22_ENGINE_LIFT_SRC} ${NAMCO22_ENGINE_C25_SRC} ${NAMCO22_ENGINE_SND_SRC}
    ${NAMCO22_ENGINE_GL_SRC} ${NAMCO22_ENGINE_SS22_SRC} ${NAMCO22_ENGINE_UI_SRC} ${NAMCO22_ENGINE_ROMZIP_SRC} ${TW_GL_SRC}
    gen/tw_c25.c gen/tw_lifted.c)
set(TW_CPU_SRC ${TW_CORE_SRC} gen/tw_snd_driver.c)

add_executable(tw ${TW_CPU_SRC})
target_compile_options(tw PRIVATE -Wall)
target_link_libraries(tw ${SDL2_LIBRARIES} ${TW_GL_LIB} ZLIB::ZLIB m)

# Development builds (the instruction trace and the translation gate): only when their sources are present -- the public tree
# has just the game.
if(EXISTS ${NAMCO22_TOOLS_DIR}/sndoracle/sound_oracle.c)
# instruction-trace build: every instruction logs PC + D0-D7/A0-A7
add_executable(tw_trace ${TW_CPU_SRC})
target_compile_options(tw_trace PRIVATE -Wall)
target_compile_definitions(tw_trace PRIVATE RR_TRACE)
target_link_libraries(tw_trace ${SDL2_LIBRARIES} OpenGL::GL ZLIB::ZLIB m)

# THE TRANSLATION GATE / COVERAGE BUILD (dev only, never shipped -- HARD RULE 1): the same host with
# the interpreter oracle linked in. TW_C25=oracle runs the program through it (C71_COV=file records
# the addresses it executes); TW_C25=lockstep runs the translation with a shadow master on the
# oracle beside it and requires their state equal after every slice (tools/c25_gate.sh).
# (the sound program runs through tools/sndoracle's fetch/decode loop here instead of the translation:
#  SND_COV=<file> records the instructions it executes, the translator's input)
add_executable(tw_oracle ${TW_CORE_SRC} ${NAMCO22_C25_ORACLE_SRC}
    ${NAMCO22_TOOLS_DIR}/sndoracle/m37710.c ${NAMCO22_TOOLS_DIR}/sndoracle/sound_oracle.c)
target_compile_options(tw_oracle PRIVATE -Wall)
target_compile_definitions(tw_oracle PRIVATE TW_ORACLE)
target_link_libraries(tw_oracle ${SDL2_LIBRARIES} OpenGL::GL ZLIB::ZLIB m)
endif()
